cmake_minimum_required(VERSION 3.22)
project(lane_store LANGUAGES C CXX)
include(verify-dependencies.cmake)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/schema.sql" LANE_SCHEMA)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/registry.sql" LANE_REGISTRY)
configure_file(schema.h.in "${CMAKE_CURRENT_BINARY_DIR}/lane_schema.h" @ONLY)
add_library(lane_sqlite STATIC vendor/sqlite/sqlite3.c)
target_include_directories(lane_sqlite PUBLIC vendor/sqlite)
target_compile_definitions(lane_sqlite PRIVATE SQLITE_THREADSAFE=1 SQLITE_DEFAULT_FOREIGN_KEYS=1 SQLITE_DQS=0 SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_SECURE_DELETE=1)
add_library(lane_unicode STATIC vendor/utf8proc/utf8proc.c)
target_include_directories(lane_unicode PUBLIC vendor/utf8proc)
target_compile_definitions(lane_unicode PUBLIC UTF8PROC_STATIC)
add_library(lane_url STATIC vendor/ada/ada.cpp)
target_include_directories(lane_url PUBLIC vendor/ada)
target_compile_features(lane_url PUBLIC cxx_std_17)
add_library(lane_store STATIC lane_store.cc)
target_compile_features(lane_store PUBLIC cxx_std_17)
target_include_directories(lane_store PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}" PRIVATE "${CMAKE_CURRENT_BINARY_DIR}" vendor/json vendor/picosha2)
target_link_libraries(lane_store PRIVATE lane_sqlite lane_unicode lane_url)
set_target_properties(lane_sqlite lane_unicode lane_url lane_store PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(UNIX)
  find_package(Threads REQUIRED)
  target_link_libraries(lane_sqlite PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
option(LANE_STORE_TESTS "Build native engine contract tests" OFF)
if(LANE_STORE_TESTS)
  enable_testing()
  add_executable(lane_store_tests tests/store_test.cc)
  target_include_directories(lane_store_tests PRIVATE vendor/json vendor/sqlite)
  target_link_libraries(lane_store_tests PRIVATE lane_store lane_sqlite)
  set(LANE_STORE_FIXTURE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../protos/lane/v1/fixtures" CACHE PATH "Shared Lane protocol conformance fixtures")
  if(EXISTS "${LANE_STORE_FIXTURE_DIR}/tasks.json")
    add_test(NAME lane_store_contract COMMAND lane_store_tests "${LANE_STORE_FIXTURE_DIR}")
  else()
    add_test(NAME lane_store_contract COMMAND lane_store_tests)
    message(STATUS "Shared protocol fixtures absent; provide LANE_STORE_FIXTURE_DIR to run cross-language vectors")
  endif()
endif()
if(LANE_STORE_TESTS)
  add_executable(lane_store_driver tests/store_driver.cc)
  target_link_libraries(lane_store_driver PRIVATE lane_store)
endif()

cmake_minimum_required(VERSION 3.24)
project(LaneWindowsHostTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(lane_windows_policy_test policy_test.cc)
target_compile_features(lane_windows_policy_test PRIVATE cxx_std_17)
target_link_libraries(lane_windows_policy_test PRIVATE Threads::Threads)
add_test(NAME lane_windows_policy COMMAND lane_windows_policy_test)
if(MSVC)
  target_compile_options(lane_windows_policy_test PRIVATE /UNDEBUG)
else()
  target_compile_options(lane_windows_policy_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()
if(WIN32)
  add_executable(lane_windows_platform_test platform_test.cc)
  set_target_properties(lane_windows_platform_test PROPERTIES OUTPUT_NAME "lane windows platform test")
  target_compile_features(lane_windows_platform_test PRIVATE cxx_std_17)
  target_compile_definitions(lane_windows_platform_test PRIVATE UNICODE _UNICODE _WIN32_WINNT=0x0A00)
  target_link_libraries(lane_windows_platform_test PRIVATE windowsapp runtimeobject ole32 shell32 advapi32 bcrypt)
  if(MSVC)
    target_compile_options(lane_windows_platform_test PRIVATE /UNDEBUG /utf-8 /EHsc /permissive-)
  endif()
  add_test(NAME lane_windows_platform COMMAND lane_windows_platform_test)
endif()

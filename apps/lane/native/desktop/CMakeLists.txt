cmake_minimum_required(VERSION 3.24)
project(Lane VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(LANE_NODE_API_INCLUDE "" CACHE PATH "Pinned node-api-headers include directory")
if(NOT EXISTS "${LANE_NODE_API_INCLUDE}/node_api.h")
  message(FATAL_ERROR "Use pnpm build:desktop to resolve Node-API headers")
endif()
add_subdirectory(../store "${CMAKE_CURRENT_BINARY_DIR}/store")
add_library(Lane MODULE addon.cc)
set_target_properties(Lane PROPERTIES PREFIX "" SUFFIX ".node" OUTPUT_NAME "lane_host")
target_include_directories(Lane PRIVATE "${LANE_NODE_API_INCLUDE}")
target_compile_definitions(Lane PRIVATE NAPI_VERSION=8)
target_link_libraries(Lane PRIVATE lane_store)
option(LANE_NATIVE_TESTING "Allow explicit synthetic native test roots" OFF)
if(LANE_NATIVE_TESTING)
  target_compile_definitions(Lane PRIVATE LANE_NATIVE_TESTING)
endif()
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OBJCXX_STANDARD 17)
  set(CMAKE_OBJCXX_STANDARD_REQUIRED ON)
  target_sources(Lane PRIVATE macos/addon.mm macos/runtime.mm)
  target_compile_options(Lane PRIVATE "$<$<COMPILE_LANGUAGE:OBJCXX>:-fobjc-arc>")
  target_link_options(Lane PRIVATE "-undefined" "dynamic_lookup")
  target_link_libraries(Lane PRIVATE "-framework Cocoa" "-framework UserNotifications")
  if(LANE_NATIVE_TESTING)
    enable_testing()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(LaneReauthTest macos/reauth_test.mm macos/runtime.mm)
    target_compile_options(LaneReauthTest PRIVATE -fobjc-arc)
    target_compile_definitions(LaneReauthTest PRIVATE LANE_NATIVE_TESTING)
    target_link_libraries(LaneReauthTest PRIVATE lane_store "-framework Cocoa" "-framework UserNotifications")
    add_test(NAME lane_macos_reauthentication COMMAND "${Python3_EXECUTABLE}"
      "${CMAKE_CURRENT_SOURCE_DIR}/macos/run_reauth_test.py" "$<TARGET_FILE:LaneReauthTest>")
  endif()
elseif(WIN32)
  add_executable(LaneIcon windows/icon.cc)
  target_compile_definitions(LaneIcon PRIVATE UNICODE _UNICODE)
  target_sources(Lane PRIVATE windows.cc)
  set(LANE_NODE_IMPORT_LIBRARY "" CACHE FILEPATH "Pinned Lynxtron runtime import library")
  if(NOT EXISTS "${LANE_NODE_IMPORT_LIBRARY}")
    message(FATAL_ERROR "Use pnpm build:desktop to resolve the Lynxtron import library")
  endif()
  target_link_libraries(Lane PRIVATE "${LANE_NODE_IMPORT_LIBRARY}")
  include("${CMAKE_CURRENT_SOURCE_DIR}/windows/integration.cmake")
else()
  message(FATAL_ERROR "Native Lane desktop supports macOS and Windows; use web on Linux")
endif()
